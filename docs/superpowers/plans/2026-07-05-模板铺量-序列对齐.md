# 模板铺量·序列对齐升级 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 模板铺量模式一在源/目标 Excel 行不对齐（增行/删行/改名）时仍能准确铺量，并识别目标表新增工程量行、推荐项目内定额、可写入。

**Architecture:** 整表快照（一次 COM UsedRange 读取）→ 行指纹（不截断全文）→ 章节锚点分段 + 段内 Needleman-Wunsch 对齐 + 改名互撮 → 平移取数表达式行号复用既有预览/写入管线；新增行整行复制项目内同编号定额。快照失败自动回退 v1 位置寻址。

**Tech Stack:** C# 5 / WinForms（RecoExpandPanel.dll 插件），PowerShell 5 反射冒烟测试。设计文档：`docs/superpowers/specs/2026-07-05-模板铺量-序列对齐-design.md`。

```csharp
            // —— 模板铺量对齐:行指纹 ——
            // 与 GetQuantityNameForCell 的区别:给机器算相似度用,取数量列前全部非纯数字
            // 单元格全文拼接,不做 15字/3段 截断;合并区域只取一次锚点文本。
            private Dictionary<long, AiExcelCell> alignCellByRowCol;

            private static long AlignCellKey(int row, int column)
            {
                return ((long)row << 20) | (uint)column;
            }

            private void EnsureAlignCellIndex()
            {
                if (alignCellByRowCol != null)
                {
                    return;
                }

                alignCellByRowCol = new Dictionary<long, AiExcelCell>();
                foreach (AiExcelCell cell in Cells)
                {
                    if (cell != null)
                    {
                        alignCellByRowCol[AlignCellKey(cell.Row, cell.Column)] = cell;
                    }
                }
            }

            public string GetRowAlignText(int row, int maxColumnExclusive)
            {
                EnsureAlignCellIndex();
                StringBuilder sb = new StringBuilder();
                HashSet<string> usedRegions = new HashSet<string>(StringComparer.Ordinal);
                for (int col = 1; col < maxColumnExclusive; col++)
                {
                    string text = null;
                    AiExcelCell cell;
                    if (alignCellByRowCol.TryGetValue(AlignCellKey(row, col), out cell) &&
                        !cell.IsNumber && !String.IsNullOrWhiteSpace(cell.Text))
                    {
                        AiMergedRange direct = FindMergedRange(col, row);
                        if (direct != null && !usedRegions.Add(direct.Key))
                        {
                            continue;
                        }

                        text = cell.Text;
                    }
                    else
                    {
                        AiMergedRange range = FindMergedRange(col, row);
                        if (range == null || !usedRegions.Add(range.Key))
                        {
                            continue;
                        }

                        AiExcelCell anchor;
                        if (!alignCellByRowCol.TryGetValue(AlignCellKey(range.ValueRow, range.ValueColumn), out anchor) ||
                            anchor.IsNumber || String.IsNullOrWhiteSpace(anchor.Text))
                        {
                            continue;
                        }

                        text = anchor.Text;
                    }

                    if (sb.Length > 0)
                    {
                        sb.Append(' ');
                    }

                    sb.Append(text.Trim());
                }

                return sb.ToString();
            }
```

```csharp
using System;
using System.Collections.Generic;
using System.Data.SqlClient;
using System.Drawing;
using System.Globalization;
using System.Linq;
using System.Text;
using System.Text.RegularExpressions;
using System.Windows.Forms;

namespace RecoNet
{
    public partial class FormPanel : Form
    {
        // ===================== 模板铺量·序列对齐 =====================
        // 设计:docs/superpowers/specs/2026-07-05-模板铺量-序列对齐-design.md
        // 行的身份是内容不是行号:章节锚点分段 + 段内 Needleman-Wunsch,
        // 行号只作先验;剩余行宽松互撮识别"疑似改名"。

        private const int AlignMatchMinScore = 55;   // 段内配对成立的最低相似度
        private const int AlignRenameMinScore = 35;  // 疑似改名互撮的最低相似度
        private const int AlignAnchorMinScore = 50;  // 章节锚点行配对最低相似度
        private const int AlignGapPenalty = 40;      // NW 空位罚分
        private const int AlignMatchBaseline = 50;   // NW 配对基准(低于此配对不如空位)

        private sealed class AlignRowInfo
        {
            public int Row;          // Excel 行号(测试缝里=下标+1)
            public string RawText;   // 指纹原文
            public string NormText;  // 归一化后
            public bool IsAnchor;    // 章节锚点行("一、/(一)/第X章")
            public bool HasQuantity; // 数量列有数字(新增行候选的必要条件)
        }

        private sealed class AlignPair
        {
            public int SourceIndex;
            public int TargetIndex;
            public int Score;
            public bool Rename;      // 由宽松互撮配上的=疑似改名
        }

        // 归一化:全角转半角、去空白与标点、小写化;文字里嵌的数字保留(500m 的 500 是身份)。
        internal static string NormalizeAlignText(string text)
        {
            if (String.IsNullOrEmpty(text))
            {
                return "";
            }

            StringBuilder sb = new StringBuilder(text.Length);
            for (int i = 0; i < text.Length; i++)
            {
                char c = text[i];
                if (c == '　')
                {
                    continue;
                }

                if (c >= '！' && c <= '～')
                {
                    c = (char)(c - 0xFEE0);
                }

                if (Char.IsWhiteSpace(c) || !Char.IsLetterOrDigit(c))
                {
                    continue;
                }

                sb.Append(Char.ToLowerInvariant(c));
            }

            return sb.ToString();
        }

        internal static List<string> ExtractAlignNumbers(string normText)
        {
            List<string> result = new List<string>();
            if (String.IsNullOrEmpty(normText))
            {
                return result;
            }

            int i = 0;
            while (i < normText.Length)
            {
                if (Char.IsDigit(normText[i]))
                {
                    int start = i;
                    while (i < normText.Length && (Char.IsDigit(normText[i]) || normText[i] == '.'))
                    {
                        i++;
                    }

                    result.Add(normText.Substring(start, i - start).TrimEnd('.'));
                }
                else
                {
                    i++;
                }
            }

            return result;
        }

        private static HashSet<string> BuildAlignBigrams(string text)
        {
            HashSet<string> grams = new HashSet<string>(StringComparer.Ordinal);
            if (text.Length == 1)
            {
                grams.Add(text);
                return grams;
            }

            for (int i = 0; i + 1 < text.Length; i++)
            {
                grams.Add(text.Substring(i, 2));
            }

            return grams;
        }

        // 相似度 0-100:字符 bigram Dice;双方都含数字且数字集不相交时重罚(/3)。
        internal static int AlignNameScore(string leftNorm, string rightNorm)
        {
            if (String.IsNullOrEmpty(leftNorm) || String.IsNullOrEmpty(rightNorm))
            {
                return 0;
            }

            if (String.Equals(leftNorm, rightNorm, StringComparison.Ordinal))
            {
                return 100;
            }

            HashSet<string> leftGrams = BuildAlignBigrams(leftNorm);
            HashSet<string> rightGrams = BuildAlignBigrams(rightNorm);
            if (leftGrams.Count == 0 || rightGrams.Count == 0)
            {
                return 0;
            }

            int common = leftGrams.Count(g => rightGrams.Contains(g));
            int score = (int)Math.Round(200.0 * common / (leftGrams.Count + rightGrams.Count));
            List<string> leftNums = ExtractAlignNumbers(leftNorm);
            List<string> rightNums = ExtractAlignNumbers(rightNorm);
            if (leftNums.Count > 0 && rightNums.Count > 0 && !leftNums.Any(n => rightNums.Contains(n)))
            {
                score /= 3;
            }

            return score > 100 ? 100 : score;
        }

        private static readonly Regex ChapterAnchorRegex = new Regex(
            "^\\s*(第?[一二三四五六七八九十百]+[、.．章节部分]|[(（][一二三四五六七八九十]+[)）])",
            RegexOptions.Compiled);

        // 章节锚点行:"一、/(一)/第X章/第X部分"等。用原文判断(归一化会去掉顿号括号)。
        internal static bool IsChapterAnchorText(string rawText)
        {
            if (String.IsNullOrEmpty(rawText))
            {
                return false;
            }

            return ChapterAnchorRegex.IsMatch(rawText.Trim());
        }

        // 通用 NW 全局对齐:得分>=minScore 的配对写入 pairs(下标为各自列表下标)。
        private static void AlignByNeedlemanWunsch(List<string> left, List<string> right, int minScore, List<AlignPair> pairs)
        {
            int n = left.Count;
            int m = right.Count;
            if (n == 0 || m == 0)
            {
                return;
            }

            int[,] dp = new int[n + 1, m + 1];
            for (int i = 1; i <= n; i++)
            {
                dp[i, 0] = dp[i - 1, 0] - AlignGapPenalty;
            }

            for (int j = 1; j <= m; j++)
            {
                dp[0, j] = dp[0, j - 1] - AlignGapPenalty;
            }

            for (int i = 1; i <= n; i++)
            {
                for (int j = 1; j <= m; j++)
                {
                    int match = dp[i - 1, j - 1] + AlignNameScore(left[i - 1], right[j - 1]) - AlignMatchBaseline;
                    int del = dp[i - 1, j] - AlignGapPenalty;
                    int ins = dp[i, j - 1] - AlignGapPenalty;
                    dp[i, j] = Math.Max(match, Math.Max(del, ins));
                }
            }

            int ci = n;
            int cj = m;
            while (ci > 0 && cj > 0)
            {
                int score = AlignNameScore(left[ci - 1], right[cj - 1]);
                if (dp[ci, cj] == dp[ci - 1, cj - 1] + score - AlignMatchBaseline)
                {
                    if (score >= minScore)
                    {
                        pairs.Add(new AlignPair { SourceIndex = ci - 1, TargetIndex = cj - 1, Score = score });
                    }

                    ci--;
                    cj--;
                }
                else if (dp[ci, cj] == dp[ci - 1, cj] - AlignGapPenalty)
                {
                    ci--;
                }
                else
                {
                    cj--;
                }
            }
        }

        // 段内对齐:[srcFrom,srcTo) × [tgtFrom,tgtTo),锚点行不参与;
        // 配对后剩余行做宽松互撮(疑似改名,贪心取最高分)。
        private static void AlignSegment(List<AlignRowInfo> src, int srcFrom, int srcTo,
            List<AlignRowInfo> tgt, int tgtFrom, int tgtTo, List<AlignPair> pairs)
        {
            List<int> srcIdx = new List<int>();
            List<int> tgtIdx = new List<int>();
            for (int i = srcFrom; i < srcTo; i++)
            {
                if (!src[i].IsAnchor)
                {
                    srcIdx.Add(i);
                }
            }

            for (int j = tgtFrom; j < tgtTo; j++)
            {
                if (!tgt[j].IsAnchor)
                {
                    tgtIdx.Add(j);
                }
            }

            if (srcIdx.Count == 0 || tgtIdx.Count == 0)
            {
                return;
            }

            List<AlignPair> local = new List<AlignPair>();
            AlignByNeedlemanWunsch(
                srcIdx.Select(i => src[i].NormText).ToList(),
                tgtIdx.Select(j => tgt[j].NormText).ToList(),
                AlignMatchMinScore, local);

            HashSet<int> usedSrc = new HashSet<int>();
            HashSet<int> usedTgt = new HashSet<int>();
            foreach (AlignPair p in local)
            {
                pairs.Add(new AlignPair { SourceIndex = srcIdx[p.SourceIndex], TargetIndex = tgtIdx[p.TargetIndex], Score = p.Score });
                usedSrc.Add(p.SourceIndex);
                usedTgt.Add(p.TargetIndex);
            }

            List<int> leftoverSrc = new List<int>();
            List<int> leftoverTgt = new List<int>();
            for (int i = 0; i < srcIdx.Count; i++)
            {
                if (!usedSrc.Contains(i))
                {
                    leftoverSrc.Add(i);
                }
            }

            for (int j = 0; j < tgtIdx.Count; j++)
            {
                if (!usedTgt.Contains(j))
                {
                    leftoverTgt.Add(j);
                }
            }

            while (leftoverSrc.Count > 0 && leftoverTgt.Count > 0)
            {
                int bestScore = -1;
                int bestSi = -1;
                int bestTj = -1;
                foreach (int si in leftoverSrc)
                {
                    foreach (int tj in leftoverTgt)
                    {
                        int score = AlignNameScore(src[srcIdx[si]].NormText, tgt[tgtIdx[tj]].NormText);
                        if (score > bestScore)
                        {
                            bestScore = score;
                            bestSi = si;
                            bestTj = tj;
                        }
                    }
                }

                if (bestScore < AlignRenameMinScore)
                {
                    break;
                }

                pairs.Add(new AlignPair { SourceIndex = srcIdx[bestSi], TargetIndex = tgtIdx[bestTj], Score = bestScore, Rename = true });
                leftoverSrc.Remove(bestSi);
                leftoverTgt.Remove(bestTj);
            }
        }

        // 总编排:先对齐锚点序列切段,再逐段对齐条目行。
        private static List<AlignPair> AlignCore(List<AlignRowInfo> src, List<AlignRowInfo> tgt)
        {
            List<AlignPair> pairs = new List<AlignPair>();
            List<int> srcAnchorIdx = new List<int>();
            List<int> tgtAnchorIdx = new List<int>();
            for (int i = 0; i < src.Count; i++)
            {
                if (src[i].IsAnchor)
                {
                    srcAnchorIdx.Add(i);
                }
            }

            for (int j = 0; j < tgt.Count; j++)
            {
                if (tgt[j].IsAnchor)
                {
                    tgtAnchorIdx.Add(j);
                }
            }

            List<AlignPair> anchorPairs = new List<AlignPair>();
            AlignByNeedlemanWunsch(
                srcAnchorIdx.Select(i => src[i].NormText).ToList(),
                tgtAnchorIdx.Select(j => tgt[j].NormText).ToList(),
                AlignAnchorMinScore, anchorPairs);

            int prevSrc = 0;
            int prevTgt = 0;
            foreach (AlignPair ap in anchorPairs.OrderBy(p => p.SourceIndex))
            {
                int srcCut = srcAnchorIdx[ap.SourceIndex];
                int tgtCut = tgtAnchorIdx[ap.TargetIndex];
                AlignSegment(src, prevSrc, srcCut, tgt, prevTgt, tgtCut, pairs);
                pairs.Add(new AlignPair { SourceIndex = srcCut, TargetIndex = tgtCut, Score = ap.Score });
                prevSrc = srcCut + 1;
                prevTgt = tgtCut + 1;
            }

            AlignSegment(src, prevSrc, src.Count, tgt, prevTgt, tgt.Count, pairs);
            return pairs;
        }

        private static List<AlignRowInfo> BuildAlignInfosFromTexts(List<string> texts)
        {
            List<AlignRowInfo> infos = new List<AlignRowInfo>();
            for (int i = 0; i < texts.Count; i++)
            {
                AlignRowInfo info = new AlignRowInfo();
                info.Row = i + 1;
                info.RawText = texts[i] ?? "";
                info.NormText = NormalizeAlignText(info.RawText);
                info.IsAnchor = IsChapterAnchorText(info.RawText);
                infos.Add(info);
            }

            return infos;
        }

        // 测试缝:纯文本序列对齐。返回长度=source.Count 的数组,
        // 值=匹配到的 target 下标(0基),未匹配=-1。
        internal static int[] AlignTextRows(List<string> sourceTexts, List<string> targetTexts)
        {
            List<AlignRowInfo> src = BuildAlignInfosFromTexts(sourceTexts);
            List<AlignRowInfo> tgt = BuildAlignInfosFromTexts(targetTexts);
            int[] map = new int[src.Count];
            for (int i = 0; i < map.Length; i++)
            {
                map[i] = -1;
            }

            foreach (AlignPair p in AlignCore(src, tgt))
            {
                map[p.SourceIndex] = p.TargetIndex;
            }

            return map;
        }
    }
```

```csharp
        private static List<AlignRowInfo> BuildAlignRowInfos(AiExcelSelectionContext ctx, int qtyColumn)
        {
            HashSet<int> rows = new HashSet<int>();
            HashSet<int> qtyRows = new HashSet<int>();
            foreach (AiExcelCell cell in ctx.Cells)
            {

                {

                }

                if (cell.Column < qtyColumn && !cell.IsNumber && !String.IsNullOrWhiteSpace(cell.Text))
                {
                    rows.Add(cell.Row);
                }

                if (cell.Column == qtyColumn && cell.IsNumber)
                {
                    rows.Add(cell.Row);
                    qtyRows.Add(cell.Row);
                }
            }

        private static bool IsSummaryRowText(string normText)
        {
            if (String.IsNullOrEmpty(normText))
            {
                return false;
            }

            return normText.StartsWith("合计", StringComparison.Ordinal)
                || normText.StartsWith("小计", StringComparison.Ordinal)
                || normText.StartsWith("总计", StringComparison.Ordinal);
        }

        private sealed class RowAlignmentResult
        {
            public readonly Dictionary<int, int> SourceRowToTargetRow = new Dictionary<int, int>();   // Excel 行号映射
            public readonly HashSet<int> RenameSourceRows = new HashSet<int>();                        // 疑似改名的源行
            public readonly List<AlignRowInfo> NewTargetRows = new List<AlignRowInfo>();               // 目标表新增行(带数量,排除锚点/合计)

        private static RowAlignmentResult BuildRowAlignment(AiExcelSelectionContext sourceCtx, int sourceQtyColumn,
            AiExcelSelectionContext targetCtx, int targetQtyColumn)
        {
            List<AlignRowInfo> src = BuildAlignRowInfos(sourceCtx, sourceQtyColumn);
            List<AlignRowInfo> tgt = BuildAlignRowInfos(targetCtx, targetQtyColumn);
            RowAlignmentResult result = new RowAlignmentResult();
            HashSet<int> usedTargetIdx = new HashSet<int>();
            foreach (AlignPair p in AlignCore(src, tgt))
            {
                result.SourceRowToTargetRow[src[p.SourceIndex].Row] = tgt[p.TargetIndex].Row;
                if (p.Rename)
                {
                    result.RenameSourceRows.Add(src[p.SourceIndex].Row);
                }

                usedTargetIdx.Add(p.TargetIndex);
            }

            for (int j = 0; j < tgt.Count; j++)
            {
                AlignRowInfo info = tgt[j];
                if (usedTargetIdx.Contains(j) || info.IsAnchor || !info.HasQuantity || IsSummaryRowText(info.NormText))
                {
                    continue;
                }

                result.NewTargetRows.Add(info);
            }

            return result;
        }

        private static readonly Regex ShiftCellRegex = new Regex("(\\$?)([A-Z]{1,3})(\\$?)(\\d+)", RegexOptions.Compiled);

        // 表达式行号整体平移:E4+E5 平移2行 => E6+E7。用正则逐格替换,避免字符串 Replace 串键。
        internal static string ShiftExprRows(string expr, int rowDelta)
        {
            if (String.IsNullOrWhiteSpace(expr) || rowDelta == 0)
            {
                return expr;
            }

            return ShiftCellRegex.Replace(expr.ToUpperInvariant(), delegate(Match m)
            {
                int row;
                if (!Int32.TryParse(m.Groups[4].Value, out row))
                {
                    return m.Value;
                }

                int shifted = row + rowDelta;
                if (shifted < 1)
                {
                    return m.Value;
                }

                return m.Groups[1].Value + m.Groups[2].Value + m.Groups[3].Value + shifted.ToString(CultureInfo.InvariantCulture);
            });
        }
```

```csharp
        private static List<NewRowQuotaOption> LoadProjectQuotaOptions(Form mainForm)
        {
            List<NewRowQuotaOption> options = new List<NewRowQuotaOption>();
            try
            {
                using (SqlConnection conn = AgentCreateWorkConnection(mainForm))

                            "where 定额编号 is not null and ltrim(rtrim(定额编号)) <> '' and 定额编号 <> '-' " +

                        using (SqlDataReader reader = cmd.ExecuteReader())
                        {
                            while (reader.Read())
                            {
                                NewRowQuotaOption opt = new NewRowQuotaOption();
                                opt.Code = reader.IsDBNull(0) ? "" : Convert.ToString(reader.GetValue(0)).Trim();
                                opt.Name = reader.IsDBNull(1) ? "" : Convert.ToString(reader.GetValue(1)).Trim();
                                opt.Unit = reader.IsDBNull(2) ? "" : Convert.ToString(reader.GetValue(2)).Trim();

        private static List<NewRowQuotaOption> RankNewRowOptions(List<NewRowQuotaOption> all, string rowName)
```
